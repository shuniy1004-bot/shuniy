    var esc = SN.esc, txt = SN.txt;
    function arr(v, fb) {
      if (Array.isArray(v)) return v;
      if (typeof v === 'string' && v.trim()) return v.split(/\n|,/).map(function (s) { return s.trim(); }).filter(Boolean);
      return fb;
    }
    function setHook(k, v) {
      document.querySelectorAll('[data-hook="' + k + '"]').forEach(function (el) {
        if (v != null && txt(v).trim() !== '') el.innerHTML = esc(v).replace(/\n/g, '<br>');
      });
    }
    function chips(id, list, cls) {
      var el = document.getElementById(id); if (!el) return;
      el.innerHTML = (list || []).map(function (x) {
        return '<span class="sn-chip' + (cls ? ' ' + cls : '') + '">' + esc(x) + '</span>';
      }).join('') || '<span style="color:var(--tx-dim);font-size:12px">—</span>';
    }
    function renderTmi(list) {
      var el = document.getElementById('tmi'); if (!el) return;
      el.innerHTML = (list || []).map(function (t) { return '<li>' + esc(t) + '</li>'; }).join('')
        || '<li style="color:var(--tx-dim)">아직 등록된 내용이 없습니다</li>';
    }
    function renderStats(list) {
      var el = document.getElementById('stats'); if (!el) return;
      var rows = (list || []).map(function (s) {
        var label = txt(s && s.label != null ? s.label : s);
        var val = parseInt(s && s.value != null ? s.value : 0, 10) || 0;
        val = Math.max(0, Math.min(100, val));
        return '<div class="stat-item"><span class="sl">' + esc(label) + '</span><span class="sv">' + val + '</span>' +
               '<span class="bar"><span class="fl" data-w="' + val + '"></span></span></div>';
      }).join('');
      el.innerHTML = rows || '<div class="sn-empty">아직 등록된 능력치가 없습니다</div>';
      /* fill directly after a re-render — the reveal observer only fires once */
      setTimeout(function () {
        el.querySelectorAll('.fl[data-w]').forEach(function (f) { f.style.width = f.dataset.w + '%'; });
      }, 120);
    }
    function renderMilestones(list) {
      var el = document.getElementById('milestones'); if (!el) return;
      var rows = (list || []).map(function (m) {
        var state, text;
        if (typeof m === 'string') { var p = m.split('|'); state = p[0] || ''; text = p[1] || ''; }
        else { m = m || {}; state = txt(m.state); text = txt(m.text); }
        var done = /완료|달성|done/i.test(state);
        return '<div class="mile-row' + (done ? ' done' : '') + '"><span class="st">' + esc(state || '진행') + '</span>' +
               '<span class="tx">' + esc(text) + '</span></div>';
      }).join('');
      el.innerHTML = rows || '<div class="sn-empty">아직 등록된 목표가 없습니다</div>';
    }
    function renderWeek(week) {
      var el = document.getElementById('week'); if (!el) return;
      var D = ['월', '화', '수', '목', '금', '토', '일'];
      var jsDay = new Date().getDay();          /* JS: 0=일 -> our index: 0=월 */
      var todayIdx = (jsDay + 6) % 7;
      el.innerHTML = D.map(function (d, i) {
        var t = txt(week && week[i] != null ? week[i] : '').trim();
        var off = !t || t === '휴방' || t === '-';
        return '<div class="week-cell' + (off ? ' off' : '') + (i === todayIdx ? ' today' : '') + '">' +
               '<span class="d">' + d + '</span><span class="t">' + (off ? '휴방' : esc(t)) + '</span></div>';
      }).join('');
    }
    function renderContents(list) {
      var el = document.getElementById('contents'); if (!el) return;
      var rows = (list || []).map(function (c) {
        c = c || {};
        var link = txt(c.url).trim();
        return '<div class="filmo-row"><span class="dt">' + esc(c.date || '') + '</span>' +
               '<span class="tt">' + esc(c.title || '') + '</span>' +
               (link ? '<a class="go" href="' + esc(link) + '" target="_blank" rel="noreferrer">WATCH ↗</a>' : '<span></span>') +
               '</div>';
      }).join('');
      el.innerHTML = rows || '<div class="sn-empty">아직 등록된 컨텐츠가 없습니다</div>';
    }
    function renderLinks(L) {

      L = L || {};

      el.innerHTML = defs.filter(function (d) { return txt(d[2]).trim() !== ''; }).map(function (d) {

      }).join('');
      var lv = document.getElementById('live-link');

    /* Replay grid — original_songs(vod_id, title) filled from admin */
    function renderVod(rows) {
      var el = document.getElementById('ovod-grid'); if (!el) return;
      if (!rows || !rows.length) {
        el.innerHTML = '<div class="vod-ph">등록된 다시보기가 없습니다</div>' +
                       '<div class="vod-ph">등록된 다시보기가 없습니다</div>' +
                       '<div class="vod-ph">등록된 다시보기가 없습니다</div>';
        return;
      }
      el.innerHTML = rows.map(function (r) {
        var id = txt(r.vod_id).trim();
        if (!id) return '';
        return '<div><div class="vod-box"><iframe src="https://vod.sooplive.co.kr/player/' + esc(id) +
               '/embed" allowfullscreen loading="lazy" title="' + esc(r.title || '다시보기') + '"></iframe></div>' +
               '<p class="vod-cap">' + esc(r.title || '') + '</p></div>';
      }).join('');
    }

    async function load() {
      try {
        var res = await db.from('profile').select('data').eq('id', 1);
        var d = res && res.data && res.data[0] && res.data[0].data;
        applyAll(d || {});
      } catch (e) { applyAll({}); }
      try {
        var songs = await fetchAll('original_songs', { order: 'sort_order', asc: true });
        renderVod(songs);
      } catch (e) { renderVod([]); }
      finally { SN.dataReady(); SN.reveal(); }
    }

    applyAll({});        /* render defaults immediately, then let the DB overwrite */
    renderVod([]);
    load();

        SN.bump(wrap);
        var r = wrap.getBoundingClientRect();
        SN.pop(r.left + r.width / 2, r.top + r.height / 2, 6);
      };
      wrap.addEventListener('click', tap);
      wrap.addEventListener('keydown', function (e) { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); tap(); } });
    }
